```java
/** 按类型更新状态；状态必须在该类型的合法词表内，否则抛 IllegalArgumentException。 */
public boolean updateStatus(String type, long id, String status) {
    Set<String> allowed = switch (type) {
        case "internship" -> Set.of("new", "applied", "favorite", "ignore");
        case "assignment", "todo" -> Set.of("pending", "done");
        default -> throw new IllegalArgumentException("不支持标记的类型: " + type);
    };
    if (status == null || !allowed.contains(status)) {
        throw new IllegalArgumentException("类型 " + type + " 不支持状态: " + status);
    }
    return switch (type) {
        case "internship" -> internshipMapper.update(null,
                new LambdaUpdateWrapper<Internship>().eq(Internship::getId, id)
                        .set(Internship::getStatus, status)) == 1;
        case "assignment" -> assignmentMapper.update(null,
                new LambdaUpdateWrapper<Assignment>().eq(Assignment::getId, id)
                        .set(Assignment::getStatus, status)) == 1;
        case "todo" -> todoMapper.update(null,
                new LambdaUpdateWrapper<Todo>().eq(Todo::getId, id)
                        .set(Todo::getStatus, status)) == 1;
        default -> false;
    };
}

/** 精确去重查询（trim 后 company+position 相等即视为同一实习）。 */
public Optional<InternshipItem> findInternshipByCompanyAndPosition(String company, String position) {
    Internship found = internshipMapper.selectOne(new LambdaQueryWrapper<Internship>()
            .eq(Internship::getCompany, company == null ? "" : company.trim())
            .eq(Internship::getPosition, position == null ? "" : position.trim())
            .last("LIMIT 1"));
    return found == null ? Optional.empty() : Optional.of(toInternshipItem(found));
}
```
